cmake_minimum_required(VERSION 3.16)
project(ProgrammingPractice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(merge_sort src/merge_sort.cpp)
add_executable(character_frequency src/character_frequency.cpp)
add_executable(practice_tests tests/core_tests.cpp)
add_executable(benchmark benchmarks/benchmark.cpp)
foreach(target merge_sort character_frequency practice_tests benchmark)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
enable_testing()
add_test(NAME core_checks COMMAND practice_tests)
